"""The crawl: every portfolio repo checked by rapp-1's own rapp_check.py at the canon pin, outside the Hive.

Each repo is cloned at depth 1 into <work>/cache/clones/ (at most six at a time), checked with
`rapp_check.py <repo> --json` run from the clones folder (so the output names the repo, never a device path, and
anyone can reproduce its bytes from the two repos at their commits), measured (links to other portfolio repos,
"experimental" mentions, the README facts a network-header PR needs) and deleted. The results stay in
<work>/sweep/; the clones do not. See docs/crawler.md.
"""
from __future__ import annotations

import difflib
import functools
import json
import os
import re
import subprocess
import sys
import time
from collections import Counter
from pathlib import Path

from . import checker, lifecycle, util
from .config import Settings
from .constants import CANON_RAPP1, END, MAX_WORKERS, OWNER, PUBLIC_REPO, REPO_URL, START
from .headers import header_present
from .inventory import family

README_NAME = re.compile(r"readme(\.[A-Za-z0-9]+)?", re.I)
MARKDOWN_EXT = {".md", ".markdown", ".mdown", ".mkdn"}
LINKED_MARKDOWN = (".md", ".markdown", ".mdx")
GENERATED = re.compile(r"generated by|auto-?generated|do not edit|don't edit|autogenerated", re.I)
TEST_PATH = re.compile(r"(^|/)(tests?|spec)/|(^|/)test_[^/]*\.py$|\.test\.[mc]?[jt]s$")
_OWNER = re.escape(OWNER)
LINK_URL = re.compile(rf"(?:github\.com/{_OWNER}/|raw\.githubusercontent\.com/{_OWNER}/|{_OWNER}\.github\.io/)"
                      r"([A-Za-z0-9._-]+)", re.I)
LINK_BARE = re.compile(rf"(?<![A-Za-z0-9._/-]){_OWNER}/([A-Za-z0-9._-]+)", re.I)
LINK_USES = re.compile(rf"^\s*-?\s*(?:uses|repository)\s*:\s*['\"]?{_OWNER}/([A-Za-z0-9._-]+)", re.I | re.M)
SUBMODULE_URL = re.compile(r"^\s*url\s*=\s*(\S+)", re.I | re.M)
SUBMODULE_OWNER = re.compile(rf"(?:^|[/:@]){_OWNER}/([A-Za-z0-9._-]+)", re.I)
PIN_NAME = re.compile(r"(?i)(^|[_.-])(pin|pins|lock)([_.-]|$)")
PIN_EXT = ("json", "yaml", "yml", "toml")  # a pin file: one of these whose name says pin, pins or lock ...
LOCK_EXT = ("lock", "pin")  # ... or any *.lock or *.pin file (agent.lock, Cargo.lock, requirements.lock)
HEADER_BLOCK = re.compile(re.escape(START) + ".*?" + re.escape(END), re.S)
MAX_LINK_FILE = 4 << 20  # bytes; bigger files are not read for links

CHECKER_LABEL = f"{OWNER}/rapp-1 rapp_check.py at {CANON_RAPP1[:7]}"
EMPTY_REASON = "the repository is empty: GitHub holds no commit to clone or check"
CRAWL_FAILED = "the crawl failed on it: "  # a crawl failure (not the repo's); such a crawl writes no crawl.json
CHECK_TIMEOUT = 900  # seconds for one rapp_check run
GIT_TIMEOUT = 1800  # seconds for one clone, fetch or Git LFS pull
READ_TIMEOUT = 600  # seconds for one local git read (ls-files, grep, rev-parse)
RETRY_DELAYS = (5, 10)  # seconds to wait before the second and the third clone attempt
GIT = ("git", "-c", "core.autocrlf=false", "-c", "core.eol=lf", "-c", "core.longpaths=true",  # the same bytes on
       "-c", f"core.attributesFile={os.devnull}")  # every platform, whatever this device's own git settings say
GIT_ENV = {"GIT_LFS_SKIP_SMUDGE": "1", "GIT_TERMINAL_PROMPT": "0", "GCM_INTERACTIVE": "never"}
CHECKED_SUFFIXES = (".json", ".egg")  # what rapp_check reads (case-sensitive, as it matches them)
LFS_POINTER = b"version https://git-lfs.github.com/spec/"
HEAD_KEYS = ("artifact", "rule", "detail", "status")
INVISIBLE = re.compile("[\x00-\x1f\x7f-\x9f\u200b-\u200f\u2028-\u202e\u2060-\u206f\ufeff]")

_say = functools.partial(print, flush=True)


# ---- small pieces ----------------------------------------------------------------------------------------

def _duration(seconds: int) -> str:
    if seconds % 60:
        return f"{seconds} seconds"
    minutes = seconds // 60
    return f"{minutes} minute" + ("" if minutes == 1 else "s")


def _redact(text: str, redact=()) -> str:
    """`text` with each (device path, label) pair of `redact` applied, a path only where it stands whole (so the
    home folder "/root" never touches a repo named "rootkit")."""
    for old, new in redact:
        text = re.sub(rf"(?<![\w.-]){re.escape(old)}(?![\w-]|\.\w)", lambda _match: new, text)
    return text


def clean(text, limit: int, redact=(), last_line: bool = False) -> str:
    """Text from a tool's error output, fit for a public reason: one line (the last non-empty one if asked), device
    paths replaced (`redact` holds (path, label) pairs), no control, bidi or invisible characters, at most `limit`."""
    lines = [line for line in str(text).splitlines() if line.strip()]
    text = (lines[-1] if lines else "") if last_line else " ".join(lines)
    return " ".join(INVISIBLE.sub(" ", _redact(text, redact)).split())[:limit]


def _inside(root: Path, path: Path) -> bool:
    try:
        path.resolve().relative_to(Path(root).resolve())
    except (OSError, RuntimeError, ValueError):
        return False
    return True


def _read_text(root, rel_path: str, limit: int = MAX_LINK_FILE) -> str | None:
    """A tracked file's text (UTF-8, undecodable bytes replaced), or None for a symlink, a non-file or a big file."""
    path = Path(root) / rel_path
    try:
        if path.is_symlink() or not path.is_file() or path.stat().st_size > limit:
            return None
        return path.read_bytes().decode("utf-8", "replace")
    except OSError:
        return None


PATHSPEC_ENV = {"GIT_LITERAL_PATHSPECS": "0", "GIT_GLOB_PATHSPECS": "0", "GIT_NOGLOB_PATHSPECS": "0",
                "GIT_ICASE_PATHSPECS": "0"}  # pathspec magic (the network's exclusions) whatever the device sets


def _git_read(root, *args: str, timeout: int = READ_TIMEOUT, ok=(0,)) -> str:
    done = util.run(*GIT, "-C", str(root), *args, timeout=timeout, env=PATHSPEC_ENV)
    if done.returncode not in ok:
        raise RuntimeError(f"git {args[0]} failed (exit {done.returncode}): {clean(done.stderr, 160, last_line=True)}")
    return done.stdout


def _grep_files(root, *pattern: str, timeout: int = READ_TIMEOUT) -> list[str]:
    """The tracked text files holding a match, in path order (NUL-separated, so any file name survives)."""
    out = _git_read(root, "grep", "-l", "-z", "-I", *pattern, timeout=timeout, ok=(0, 1))
    return [p for p in out.split("\0") if p]


# ---- the README -----------------------------------------------------------------------------------------

def find_readme(root) -> str:
    """The README GitHub shows: .github/, then the root, then docs/ (a markdown one preferred). A README that
    resolves outside the checkout (a symlink out of it) is never picked, so nothing outside it is read."""
    root = Path(root)
    for sub in (".github", "", "docs"):
        folder = root / sub
        if not folder.is_dir():
            continue
        names = sorted((p.name for p in folder.iterdir()
                        if p.is_file() and README_NAME.fullmatch(p.name) and _inside(root, p)),
                       key=lambda n: (Path(n).suffix.lower() not in MARKDOWN_EXT, n))
        if names:
            return f"{sub}/{names[0]}".lstrip("/")
    return ""


def readme_facts(root, files, repo: str, apply_header=None, *, timeout: int = READ_TIMEOUT) -> dict:
    """What a header PR would change, and what could stand in its way (so the clone can go): receipts that pin the
    README by SHA-256 or git blob id, a "generated" marker, tests that read the README, and the header's diff.
    `apply_header(text, repo) -> (new_text, action)` defaults to rapp1_network.headers.apply_header."""
    root = Path(root)
    readme = find_readme(root)
    facts = {"readme": readme, "readme_markdown": Path(readme).suffix.lower() in MARKDOWN_EXT if readme else False,
             "header_present": False}
    if not readme:
        return facts
    raw = (root / readme).read_bytes()
    text = raw.decode("utf-8", errors="replace")
    facts["header_present"] = header_present(text)  # both markers, outside fenced code (an example is not one)
    if not facts["readme_markdown"]:
        return facts
    if apply_header is None:
        from .headers import apply_header
    digest, blob = util.sha256(raw), util.git_blob_id(raw)
    pinned = _grep_files(root, "-F", "-e", digest, "-e", blob, timeout=timeout)
    tests = {p for p in files if TEST_PATH.search(p)}
    readers = [p for p in _grep_files(root, "-e", "README", timeout=timeout) if p in tests] if tests else []
    try:
        new, action = apply_header(text, repo)
    except ValueError as error:
        new, action = text, f"manual: {error}"
    diff = "".join(difflib.unified_diff(text.splitlines(True), new.splitlines(True), f"a/{readme}", f"b/{readme}",
                                        n=2))
    facts.update(readme_sha256=digest, readme_receipts=[p for p in pinned if p != readme],
                 readme_generated=bool(GENERATED.search("\n".join(text.splitlines()[:15]))),
                 readme_test_readers=readers, header_action=action, header_diff=diff[:4000])
    return facts


# ---- links between portfolio repos ----------------------------------------------------------------------

def strip_header(text: str) -> str:
    """The text without the network header block(s): the network's own links are not the repo's."""
    return HEADER_BLOCK.sub("", text) if START in text else text


def submodule_repo(url: str, self_name: str) -> str | None:
    """The owner's repo a .gitmodules url names: an absolute URL, or one relative to this repo on GitHub."""
    url = url.strip("\"'")
    if url.startswith(("./", "../")):
        parts = [OWNER, self_name]
        for part in url.split("/"):
            if part == "..":
                if not parts:
                    return None
                parts.pop()
            elif part not in ("", "."):
                parts.append(part)
        return parts[1] if len(parts) == 2 and parts[0].lower() == OWNER.lower() else None
    found = SUBMODULE_OWNER.search(url)
    return found.group(1) if found else None


def _link_names(kind: str, patterns, text: str, self_name: str):
    if kind == "submodule":
        for url in SUBMODULE_URL.findall(text):
            name = submodule_repo(url, self_name)
            if name:
                yield name
        return
    for pattern in patterns:
        for match in pattern.finditer(text):
            yield match.group(1)


NETWORK_FOLDERS = ("portfolio/", "members/", "notices/")  # the network's own, in its public copy (C10)
NETWORK_CARD = ".rapp/member.md"  # the repo's card in the RAPP Hive, written from the portfolio (C11)

def crawl_links(root, files, self_name: str, known: dict) -> dict[str, set[str]]:
    """{portfolio repo: {kinds}} this checkout references. `known` maps each portfolio repo's lower-case name to
    its name; nothing else counts, and neither does the repo itself.

        markdown   links to github.com/<owner>/<repo>, <owner>.github.io/<repo>, and
                   raw.githubusercontent.com/<owner>/<repo>
        pin        pin files (*_PIN.json and the like) and lock files (*.lock): those links, and bare <owner>/<repo>
        workflow   `uses:` and `repository:` lines in .github/workflows/
        submodule  .gitmodules URLs, absolute or relative (../<repo>.git sits beside this repo on GitHub)

    The network header block is left out: its links are the network's, not the repo's. So are the network's own
    folders in its public copy (portfolio/, members/, notices/ and PUBLISHED.md in rapp-hive-public), and the
    repo's network card `.rapp/member.md` (C11: its links are the network's; its Neighbors come from this crawl)."""
    found: dict[str, set[str]] = {}
    own = self_name.lower() == PUBLIC_REPO.lower()
    for rel_path in files:
        low = rel_path.lower()
        if low == NETWORK_CARD or (own and (low.startswith(NETWORK_FOLDERS) or low == "published.md")):
            continue
        base = low.rsplit("/", 1)[-1]
        stem, _, ext = base.rpartition(".")
        kinds = []
        if low.endswith(LINKED_MARKDOWN) or README_NAME.fullmatch(base):
            kinds.append(("markdown", (LINK_URL,)))
        if (ext in PIN_EXT and PIN_NAME.search(stem)) or ext in LOCK_EXT:
            kinds.append(("pin", (LINK_URL, LINK_BARE)))
        if low.startswith(".github/workflows/") and ext in ("yml", "yaml"):
            kinds.append(("workflow", (LINK_USES,)))
        if rel_path == ".gitmodules":
            kinds.append(("submodule", ()))
        if not kinds:
            continue
        text = _read_text(root, rel_path)
        if text is None:
            continue
        text = strip_header(text)
        if OWNER not in text.lower() and rel_path != ".gitmodules":
            continue
        for kind, patterns in kinds:
            for name in _link_names(kind, patterns, text, self_name):
                name = name.rstrip(".")
                name = name[:-4] if name.lower().endswith(".git") else name
                canon = known.get(name.lower())
                if canon and canon != self_name:
                    found.setdefault(canon, set()).add(kind)
    return found


# ---- one checkout --------------------------------------------------------------------------------------

def tracked(repo_dir, timeout: int = READ_TIMEOUT) -> list[str]:
    """The checkout's tracked files, as git lists them (forward slashes, path order)."""
    return [p for p in _git_read(repo_dir, "ls-files", "-z", timeout=timeout).split("\0") if p]


def has_card(root, files, timeout: int = READ_TIMEOUT) -> bool:
    """Whether the repo carries its network card: `.rapp/member.md` tracked at the evidence commit as a regular file
    (index mode 100644 or 100755: never a link or a submodule, whatever the device's core.symlinks says) and a plain
    file inside the checkout."""
    if NETWORK_CARD not in set(files):
        return False
    entry = _git_read(root, "ls-files", "-s", "-z", "--", NETWORK_CARD, timeout=timeout).split("\0", 1)[0]
    path = Path(root) / ".rapp" / "member.md"
    return entry.split(" ", 1)[0] in ("100644", "100755") and not path.is_symlink() and path.is_file()


def experimental_mentions(root, timeout: int = READ_TIMEOUT, *, public_copy: bool = False) -> int:
    """How many times "experimental" appears (whole word, any case) in the tracked text files, leaving out the
    network's own files: the repo's card (C11) and, in the network's public copy, its own folders (C10)."""
    skip = [NETWORK_CARD, *((*NETWORK_FOLDERS, "PUBLISHED.md") if public_copy else ())]
    return len(_git_read(root, "grep", "-I", "-i", "-o", "-w", "-e", "experimental", "--", ".",
                         *(f":(exclude,icase){p}" for p in skip), timeout=timeout, ok=(0, 1)).splitlines())


def _is_lfs_pointer(path: Path) -> bool:
    try:
        if path.is_symlink() or not path.is_file() or path.stat().st_size >= 1024:
            return False
        with path.open("rb") as handle:
            return handle.read(len(LFS_POINTER)) == LFS_POINTER
    except OSError:
        return False


def lfs_unfetched(root, files, timeout: int = GIT_TIMEOUT) -> list[str]:
    """Git LFS: when any .gitattributes (at any depth) routes files through LFS, fetch the ones rapp_check reads
    (*.json, *.egg); return those still left as LFS pointers, whose real content the check cannot see."""
    root = Path(root)
    attributes = [p for p in files if p == ".gitattributes" or p.endswith("/.gitattributes")]
    if not any("filter=lfs" in (_read_text(root, p) or "") for p in attributes):
        return []
    try:
        util.run(*GIT, "-C", str(root), "lfs", "pull", "--include", "*.json,*.egg", timeout=timeout,
                 env={**GIT_ENV, "GIT_LFS_SKIP_SMUDGE": "0"})
    except subprocess.TimeoutExpired:
        pass
    return [p for p in files if p.endswith(CHECKED_SUFFIXES) and _is_lfs_pointer(root / p)]


# ---- the verdict ---------------------------------------------------------------------------------------

def parse_result(raw_text: str) -> dict:
    """rapp_check's JSON object from its raw output; {} when there is none."""
    try:
        value = json.loads(raw_text or "{}")
    except ValueError:
        return {}
    return value if isinstance(value, dict) else {}


def _findings(result: dict) -> list[dict]:
    found = result.get("findings") or []
    return [f for f in found if isinstance(f, dict)] if isinstance(found, list) else []


def reason_for(result: dict, code, raw_error: str = "", *, timeout: int = CHECK_TIMEOUT, redact=()) -> str:
    """Why a checked repo is not certified, in one line."""
    if code == "timeout":
        return f"the checker did not finish within {_duration(timeout)}, so nothing is verified"
    if "error" in result:
        return f"the checker could not read the repository: {clean(result['error'], 160, redact)}"
    findings = _findings(result)
    if not findings:
        why = clean(raw_error, 120, redact, last_line=True)
        return f"the checker stopped (exit {code}) without a verdict{': ' + why if why else ''}"
    rules = Counter(str(f.get("rule", "?")) for f in findings)
    unverified = sum(1 for f in findings if f.get("status") == "unverified")
    parts = ", ".join(f"{rule} ×{n}" if n > 1 else rule for rule, n in rules.most_common(4))
    more = f", and {len(rules) - 4} more rule(s)" if len(rules) > 4 else ""
    tail = f"; {unverified} of them unverified" if unverified else ""
    return f"{len(findings)} finding(s) from rapp_check: {parts}{more}{tail}"


def _publishable(value, redact=(), limit: int = 200) -> str:
    """A finding's text as the portfolio may show it: device paths replaced, no control, bidi or invisible
    characters, at most `limit` characters."""
    return INVISIBLE.sub(" ", _redact(str(value), redact))[:limit]


def derive(result: dict, code, raw_text: str, stderr: str = "", *, timeout: int = CHECK_TIMEOUT, redact=(),
           unverified: str = "") -> dict:
    """The record fields one rapp_check run decides: status, reason, verdict, raw_sha256, findings, findings_head
    and evidence. Pure: the same run gives the same fields. `code` is the exit code or "timeout"; `redact` holds
    (device path, label) pairs for text that may quote this device's folders; `unverified` names something the
    check could not see (it keeps the repo from being certified)."""
    result = result if isinstance(result, dict) else {}
    findings, evidence = _findings(result), result.get("evidence") or []
    passed = checker.certified(result, code)
    if passed and not unverified:
        status, reason = "certified", ""
    else:
        reasons = [unverified] if unverified else []
        if not passed:
            reasons.append(reason_for(result, code, stderr, timeout=timeout, redact=redact))
        status, reason = "not yet", "; ".join(reasons)
    return {"status": status, "reason": reason,
            "verdict": result.get("verdict") or ("timeout" if code == "timeout" else "error"),
            "raw_sha256": util.sha256(raw_text or "{}"),
            "findings": len(findings),
            "findings_head": [{k: _publishable(f.get(k, ""), redact) for k in HEAD_KEYS if f.get(k)}
                              for f in findings[:12]],
            "evidence": len(evidence) if isinstance(evidence, list) else 0}


# ---- one repo ------------------------------------------------------------------------------------------

def _path_forms(path) -> list[str]:
    """A folder as error text may spell it: as given and resolved, native, with forward slashes, and with doubled
    backslashes (Python's repr of a Windows path)."""
    forms = []
    try:
        for candidate in (Path(path), Path(path).resolve()):
            if candidate.parent != candidate:  # never a bare root such as "/"
                forms += [str(candidate), candidate.as_posix(), str(candidate).replace("\\", "\\\\")]
    except (OSError, RuntimeError):
        pass
    return forms


def redactions(settings: Settings, clone: Path, repo: str) -> tuple[tuple[str, str], ...]:
    """(device path, neutral label) pairs, longest first, for error text that may name this device's folders."""
    pairs: dict[str, str] = {}
    try:
        home = Path.home()
    except (KeyError, RuntimeError):
        home = None
    python = [(Path(p), "<python>") for p in {sys.executable, sys.prefix, sys.base_prefix} if p]
    for path, label in ((clone, repo), (settings.clones, "<clones>"), (settings.checker, "<checker>"), *python,
                        (settings.work, "<work>"), (home, "~")):
        for form in _path_forms(path) if path is not None else ():
            pairs.setdefault(form, label)
    return tuple(sorted(pairs.items(), key=lambda pair: (-len(pair[0]), pair[0])))


def git_error(stderr: str, redact=()) -> str:
    """The telling line of git's error output: its last `fatal:` or `error:` line, else its last line."""
    lines = [line for line in stderr.splitlines() if line.strip()]
    telling = [line for line in lines if line.lstrip().startswith(("fatal:", "error:"))]
    return clean((telling or lines or [""])[-1], 160, redact) or "unknown error"


def _origin(clone: Path) -> str:
    done = util.run(*GIT, "-C", str(clone), "remote", "get-url", "origin", timeout=READ_TIMEOUT)
    return done.stdout.strip() if done.returncode == 0 else ""


def _clone(clone: Path, url: str, sleep, redact=()) -> tuple[str, str]:
    """(commit, "") for a depth-1 checkout of the remote's HEAD at `clone`, or ("", error). A clone left by an
    earlier crawl of the same remote is fetched and reset instead; anything else there is deleted first."""
    error = ""
    for attempt in range(len(RETRY_DELAYS) + 1):
        if (clone / ".git").is_dir() and _origin(clone) == url:
            steps = [(*GIT, "-C", str(clone), "fetch", "-q", "--depth", "1", "--no-tags", "origin", "HEAD"),
                     (*GIT, "-C", str(clone), "reset", "-q", "--hard", "FETCH_HEAD"),
                     (*GIT, "-C", str(clone), "clean", "-qffdx")]
        else:
            util.remove_tree(clone)
            steps = [(*GIT, "clone", "-q", "--depth", "1", "--no-tags", url, str(clone))]
        error = ""
        for step in steps:
            try:
                done = util.run(*step, timeout=GIT_TIMEOUT, env=GIT_ENV)
                failed, output = done.returncode != 0, done.stderr
            except subprocess.TimeoutExpired:
                failed, output = True, f"timed out after {_duration(GIT_TIMEOUT)}"
            if failed:
                error = git_error(output, redact)
                break
        if not error:
            break
        util.remove_tree(clone)
        if attempt < len(RETRY_DELAYS):
            sleep(RETRY_DELAYS[attempt])
    if error:
        return "", error
    head = util.run(*GIT, "-C", str(clone), "rev-parse", "--verify", "-q", "HEAD", timeout=READ_TIMEOUT)
    return head.stdout.strip() if head.returncode == 0 else "", ""


def _check(settings: Settings, clone: Path, repo: str, commit: str, known: dict, apply_header, check_timeout: int,
           redact) -> dict:
    files = tracked(clone)
    unfetched = lfs_unfetched(clone, files)
    raw, code, stderr = checker.run_rapp_check(repo, Path(settings.checker).resolve(), cwd=settings.clones,
                                               timeout=check_timeout)  # absolute: it runs from the clones folder
    util.write_text(settings.sweep / f"{repo}.json", raw or "{}")
    unverified = (f"the crawl could not fetch {len(unfetched)} Git LFS file(s) that rapp_check reads "
                  f"(first: {clean(unfetched[0], 120)}), so its verdict is unverified") if unfetched else ""
    links = crawl_links(clone, files, repo, known)
    version = lifecycle.read_version_file(clone, files)
    return {**derive(parse_result(raw), code, raw, stderr, timeout=check_timeout, redact=redact,
                     unverified=unverified),
            **({"version": version, "version_source": "VERSION"} if version else {}),
            **({"member_card": True} if has_card(clone, files) else {}),
            "evidence_commit": commit,
            "experimental_mentions": experimental_mentions(clone, public_copy=repo.lower() == PUBLIC_REPO.lower()),
            "tracked_files": len(files),
            "links_to": sorted(links, key=str.lower),
            "links_kinds": {name: sorted(kinds) for name, kinds in sorted(links.items(), key=lambda kv: kv[0].lower())},
            **readme_facts(clone, files, repo, apply_header)}


def sweep_one(settings: Settings, entry: dict, today: str, known: dict, remote: str = REPO_URL, *,
              apply_header=None, sleep=time.sleep, check_timeout: int = CHECK_TIMEOUT, release=None) -> dict:
    """Clone one portfolio repo at depth 1, check it, measure it, write sweep/<repo>.json (the raw output) and
    sweep/<repo>.record.json, and delete the clone, whatever happened. The record is complete, or it is
    `unchecked` with the reason (no clone, an empty repo, or a crawl failure); never a guess. Its `version` comes
    from the root VERSION file, else from `release(repo)` (the latest GitHub release's tag, or None)."""
    repo = entry["repo"]
    base = {"repo": repo, "wave": entry["wave"], "family": entry["family"], "checked": today,
            "checker": CHECKER_LABEL, "default_branch": entry["default_branch"]}
    settings.clones.mkdir(parents=True, exist_ok=True)
    clone = settings.clones / repo
    redact = redactions(settings, clone, repo)
    try:
        try:
            commit, error = _clone(clone, f"{remote}{repo}.git", sleep, redact)
            if error:
                rec = {**base, "status": "unchecked", "reason": "the crawl could not clone it: " + error}
            elif not commit:
                rec = {**base, "status": "unchecked", "reason": EMPTY_REASON}
            else:
                rec = {**base, **_check(settings, clone, repo, commit, known, apply_header, check_timeout, redact)}
        except Exception as failure:  # one repo's failure is its record, not the crawl's end
            rec = {**base, "status": "unchecked",
                   "reason": f"{CRAWL_FAILED}{type(failure).__name__}: {clean(failure, 160, redact)}"}
        if "version" not in rec and release is not None:
            try:
                tag = lifecycle.valid_version(release(repo))
            except Exception:  # a release lookup never fails the crawl
                tag = None
            if tag:
                rec.update(version=tag, version_source="release")
        util.dump(settings.sweep / f"{repo}.record.json", rec)
        return rec
    finally:
        try:
            util.remove_tree(clone)  # keep the results, not the clones
        except OSError:
            pass  # counted as a clone left in the cache


# ---- the crawl -----------------------------------------------------------------------------------------

def sweep_args(args) -> dict:
    """sweep()'s options from command-line words: --wave 1|2|all, --repos a,b, --workers n."""
    try:
        workers = int(util.arg(args, "--workers", str(MAX_WORKERS)))
    except ValueError:
        raise SystemExit("--workers takes a number") from None
    return {"wave": util.arg(args, "--wave", "all"),
            "repos": tuple(n for n in util.arg(args, "--repos", "").split(",") if n), "workers": workers}


def sweep(settings: Settings, wave="all", repos=(), workers: int = MAX_WORKERS, remote: str = REPO_URL,
          today: str | None = None, clock=None, *, apply_header=None, say=_say, sleep=time.sleep,
          check_timeout: int = CHECK_TIMEOUT, gh=None, release=None) -> list[dict]:
    """Crawl the portfolio (or one wave, or the named repos) with rapp_check.py at the canon pin: depth-1 clones
    from `remote` + "<repo>.git", at most six at a time, each deleted after its check. Writes sweep/*.json and
    sweep/*.record.json, and data/crawl.json when the crawl covered the whole family (the only crawl a pulse may
    record, so a full crawl in which the crawl itself failed on a repo writes none and removes an older one).
    `clock()` gives the UTC time in RAPP/1's fixed form; `today` defaults to the crawl's start date. With `gh` (or a
    `release(repo)` callable), a repo without a root VERSION file gets the tag of its latest GitHub release as its
    version (paced; a failure means none)."""
    wave = str(wave)
    if wave not in ("all", "1", "2"):
        raise SystemExit(f"--wave takes 1, 2 or all, not {wave!r}")
    names = [n.strip() for n in (repos.split(",") if isinstance(repos, str) else repos) if n and n.strip()]
    everyone = family(settings)
    known = {e["repo"].lower(): e["repo"] for e in everyone}  # only public portfolio repos count as links
    unknown = [n for n in names if n.lower() not in known]
    if unknown:
        raise SystemExit(f"not in the portfolio (data/family.json): {', '.join(unknown)}")
    wanted = {known[n.lower()] for n in names}
    entries = [e for e in everyone if (wave == "all" or str(e["wave"]) == wave) and (not names or e["repo"] in wanted)]
    checker.ensure_checker(settings.checker)
    if apply_header is None:
        from .headers import apply_header  # before the first clone, so a missing module fails the crawl, not a repo
    full = wave == "all" and not names  # a full crawl: the one a pulse may record
    settings.sweep.mkdir(parents=True, exist_ok=True)
    settings.clones.mkdir(parents=True, exist_ok=True)
    if full:
        (settings.data / "crawl.json").unlink(missing_ok=True)  # until this crawl has covered every repo
    clock = clock or util.utc_now
    started_utc = clock()
    today = today or started_utc[:10]
    workers = max(1, min(int(workers), MAX_WORKERS))
    say(f"crawling {len(entries)} repo(s) with rapp_check.py at {CANON_RAPP1[:7]}: depth-1 clones, {workers} at a "
        "time, each deleted after its check")
    started, out = time.monotonic(), []
    if release is None and gh is not None:
        release = lifecycle.Releases(gh)
    one = functools.partial(sweep_one, apply_header=apply_header, sleep=sleep, check_timeout=check_timeout,
                            release=release)
    pool = cf.ThreadPoolExecutor(workers)
    try:
        for rec in pool.map(lambda e: one(settings, e, today, known, remote), entries):
            out.append(rec)
            say(f"swept {rec['repo']}: {rec['status']}" + (f" ({rec.get('verdict')})" if rec.get("verdict") else "")
                + (f" · {len(rec.get('links_to', []))} link(s)" if "links_to" in rec else "")
                + (f" · {rec['reason'][:90]}" if rec.get("reason") else ""))
    except BaseException:
        pool.shutdown(wait=True, cancel_futures=True)
        raise
    pool.shutdown(wait=True)
    left = sorted(p.name for p in settings.clones.iterdir()) if settings.clones.exists() else []
    failed = [rec["repo"] for rec in out if rec.get("reason", "").startswith(CRAWL_FAILED)]
    if full and not failed:
        util.dump(settings.data / "crawl.json", {"started_utc": started_utc, "finished_utc": clock(),
                                                 "repos": len(entries), "clones_left": len(left),
                                                 "workers": workers})
    say(f"crawl done in {time.monotonic() - started:.0f} s; clones left in the cache: {len(left)}")
    if failed:
        blocked = "; no data/crawl.json, so no pulse can record this crawl: fix the cause and crawl again"
        say(f"the crawl failed on {len(failed)} repo(s): {', '.join(failed)}" + (blocked if full else ""))
    return out
